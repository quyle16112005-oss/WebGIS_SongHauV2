// file: app.js class WebGISMonitor { constructor() { // Cấu hình cơ bản this.mapCoords = [10.045, 105.795]; this.zoomLevel = 13; this.currentMode = 'temp'; // temp, do, tss this.isScanning = true; this.measureActive = false; // Dữ liệu bộ nhớ (States) this.polygons = []; this.targets = []; this.layers = {}; // Vi chỉnh this.offsets = { lat: 0, lng: 0 }; this.initMap(); this.initCharts(); this.loadMockData(); this.bindEvents(); this.startSimulation(); } // 1. KHỞI TẠO BẢN ĐỒ initMap() { this.map = L.map('map', {zoomControl: false}).setView(this.mapCoords, this.zoomLevel); L.control.zoom({ position: 'bottomright' }).addTo(this.map); this.basemaps = { satellite: L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', { maxZoom: 18 }), osm: L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 18 }) }; this.currentBasemap = 'satellite'; this.basemaps[this.currentBasemap].addTo(this.map); // Các nhóm layer this.layers.aiGroup = L.layerGroup().addTo(this.map); this.layers.polygonGroup = L.layerGroup().addTo(this.layers.aiGroup); this.layers.targetGroup = L.layerGroup().addTo(this.layers.aiGroup); this.measureMarker = null; } // 2. KHỞI TẠO DỮ LIỆU BẮT ĐẦU (Thay vì viết thẳng vào code, cấu trúc thành hàm để sau này gọi API) loadMockData() { // Dữ liệu mô phỏng lòng sông (lấy 2 đoạn ví dụ, bạn có thể bổ sung phần còn lại) this.riverZones = [ { id: "VUNG-S1", side: "Luồng Đông", temp: 29.5, doVal: 5.6, tss: 40, coords: [[10.092, 105.736], [10.095, 105.740], [10.086, 105.752], [10.083, 105.748]] }, { id: "VUNG-S2", side: "Luồng Tây", temp: 31.2, doVal: 4.8, tss: 60, coords: [[10.064, 105.776], [10.067, 105.780], [10.057, 105.793], [10.054, 105.789]] } ]; // Dữ liệu cá/lồng bè this.targets = [ { id: "TRAM-01", type: "fish", name: "Cá Tra", lat: 10.077, lng: 105.766, baseLat: 10.077, baseLng: 105.766, speedX: 0.00012, speedY: -0.00010, dir: 1 }, { id: "TRAM-02", type: "cage", name: "Lồng bè", lat: 10.052, lng: 105.798, baseLat: 10.052, baseLng: 105.798, speedX: 0, speedY: 0, dir: 1 } ]; this.renderPolygons(); this.renderTargets(); } // Xác định màu sắc theo ngưỡng (Algorithm) getZoneColor(zone) { if (this.currentMode === 'temp') return zone.temp > 30 ? '#ff595e' : zone.temp > 28.5 ? '#ffca3a' : '#2ec4b6'; if (this.currentMode === 'do') return zone.doVal < 5.2 ? '#ff595e' : zone.doVal < 6 ? '#ffca3a' : '#2ec4b6'; return zone.tss > 55 ? '#ff595e' : zone.tss > 35 ? '#ffca3a' : '#2ec4b6'; } // 3. VẼ ĐỐI TƯỢNG LÊN BẢN ĐỒ renderPolygons() { this.layers.polygonGroup.clearLayers(); this.riverZones.forEach(zone => { // Áp dụng vi chỉnh const shiftedCoords = zone.coords.map(pt => [pt[0] + this.offsets.lat, pt[1] + this.offsets.lng]); const poly = L.polygon(shiftedCoords, { fillColor: this.getZoneColor(zone), fillOpacity: 0.6, weight: 1, color: '#fff' }).addTo(this.layers.polygonGroup); poly.bindPopup(`${zone.id}
Nhiệt độ: ${zone.temp}°C
DO: ${zone.doVal}`); }); } renderTargets() { this.layers.targetGroup.clearLayers(); const tbody = document.getElementById('table-body'); if(tbody) tbody.innerHTML = ''; this.targets.forEach(t => { const lat = t.lat + this.offsets.lat; const lng = t.lng + this.offsets.lng; const color = t.type === 'fish' ? '#ffca3a' : '#ff595e'; t.marker = L.circleMarker([lat, lng], { radius: 8, fillColor: color, weight: 2, color: '#fff', fillOpacity: 1 }) .addTo(this.layers.targetGroup) .bindPopup(`${t.name}
Tọa độ: ${lng.toFixed(4)}, ${lat.toFixed(4)}`); // Render bảng UI if(tbody) { const tr = document.createElement('tr'); tr.innerHTML = `${t.id}${t.name}${lng.toFixed(4)}, ${lat.toFixed(4)}`; tr.onclick = () => { this.map.flyTo([lat, lng], 15); t.marker.openPopup(); }; tbody.appendChild(tr); } }); } // 4. LẮNG NGHE SỰ KIỆN (DOM Events) bindEvents() { // Sự kiện click đo tự do this.map.on('click', (e) => { if (!this.measureActive) return; if (this.measureMarker) this.map.removeLayer(this.measureMarker); this.measureMarker = L.circleMarker(e.latlng, { radius: 9, fillColor: '#00f5d4', color: '#fff', weight: 3, fillOpacity: 1 }) .addTo(this.map) .bindPopup(`Đo nhanh
Tọa độ: ${e.latlng.lng.toFixed(4)}, ${e.latlng.lat.toFixed(4)}
Nhiệt độ: ${(28+Math.random()*2).toFixed(1)}°C`).openPopup(); }); // Toggle Switch AI const toggleAI = document.getElementById('toggleAI'); if(toggleAI) { toggleAI.addEventListener('change', (e) => { this.isScanning = e.target.checked; this.isScanning ? this.map.addLayer(this.layers.aiGroup) : this.map.removeLayer(this.layers.aiGroup); }); } } // 5. MÔ PHỎNG THỜI GIAN THỰC (Real-time Simulation) startSimulation() { // Cập nhật vị trí cá mỗi giây setInterval(() => { if (!this.isScanning) return; this.targets.forEach(t => { if (t.speedX === 0 && t.speedY === 0) return; t.lat += (t.speedY * t.dir); t.lng += (t.speedX * t.dir); // Giới hạn vùng bơi if (Math.abs(t.lat - t.baseLat) > 0.001) t.dir *= -1; if (t.marker) { t.marker.setLatLng([t.lat + this.offsets.lat, t.lng + this.offsets.lng]); } }); }, 1000); // Cập nhật biểu đồ mỗi 1.8s setInterval(() => { if(this.isScanning && this.charts) { const newDO = (5.4 + Math.random() * 0.8).toFixed(1); this.updateChartData(this.charts.do, newDO); } }, 1800); } // 6. XỬ LÝ BIỂU ĐỒ (Sử dụng Chart.js) initCharts() { const createChart = (canvasId, color, min, max) => { const canvas = document.getElementById(canvasId); if(!canvas) return null; return new Chart(canvas.getContext('2d'), { type: 'line', data: { labels: ['1','2','3','4','5','6'], datasets: [{ data: [5,5.5,5.2,5.8,5.4,5.6], borderColor: color }] }, options: { responsive: true, maintainAspectRatio: false, scales: { y: { min: min, max: max } } } }); }; this.charts = { do: createChart('chart1', '#2ec4b6', 4, 8) }; } updateChartData(chartInstance, newValue) { if(!chartInstance) return; chartInstance.data.datasets[0].data.shift(); chartInstance.data.datasets[0].data.push(parseFloat(newValue)); chartInstance.update(); } } // Chạy ứng dụng khi DOM tải xong document.addEventListener('DOMContentLoaded', () => { window.appGIS = new WebGISMonitor(); });
